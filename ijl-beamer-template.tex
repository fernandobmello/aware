% ── Logo overlay: bottom-right corner, frames 2 onwards ─────
%
%  \AddToShipoutPictureBG* places content in absolute page coordinates
%  on every page while it is active.  We activate it after \begin{document}
%  so it skips the title slide (frame 1).
%
%  Coordinates (x,y) are measured from the bottom-left of the page.
%  \paperwidth / \paperheight give the full page size in sp units.
%  We nudge the logo 0.25 cm from the right and 0.35 cm from the bottom.
%
\newcommand{\IJLlogoOverlay}{%
  \AddToShipoutPictureFG{%
    \AtPageLowerLeft{%
      \hspace{\dimexpr\paperwidth-3.8cm\relax}%
      \raisebox{1.4cm}{%
        \includegraphics[height=0.9cm]{IJL-largo-ingles.jpg}%
      }%
    }%
  }%
}
